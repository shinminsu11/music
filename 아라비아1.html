const tracks = [ { title: "아라비아숫자의 비밀(고요)", src: "아라비아숫자의 비밀(고요).mp3", }, { title: "아라비아숫자의 비밀", src: "아라비아숫자의 비밀.mp3", }, { title: "아라비아숫자의 비밀(여성)", src: "아라비아숫자의 비밀(여성).mp3", } ]; /* ================================================== 현재 곡 번호 ================================================== */ let currentTrack = 0; /* ================================================== 곡 선택 ================================================== */ function selectTrack(trackNumber, autoPlay = true) { const index = trackNumber - 1; if (index < 0 || index >= tracks.length) { return; } currentTrack = index; const track = tracks[index]; const audioPlayer = document.getElementById("audio-player"); const audioSource = document.getElementById("audio-source"); const currentTitle = document.getElementById("current-title"); const lyricsDisplay = document.getElementById("lyrics-display"); /* 제목 */ currentTitle.innerText = track.title; /* 음악 */ audioSource.src = track.src; /* 가사 */ lyricsDisplay.innerText = track.lyrics; /* 버튼 색상 */ document .querySelectorAll(".track-btn") .forEach(function(button, i) { button.classList.toggle( "active", i === index ); }); /* 음악 다시 불러오기 */ audioPlayer.load(); /* 직접 선택했을 때 재생 */ if (autoPlay) { audioPlayer.play().catch(function() { /* 스마트폰 브라우저가 자동재생을 막는 경우입니다. 사용자가 재생 버튼을 누르면 정상적으로 시작됩니다. */ }); } } /* ================================================== ★★★ 무한 반복 ★★★ 1 → 2 → 3 → 1 → 2 → 3 → 계속 ================================================== */ const audioPlayer = document.getElementById("audio-player"); audioPlayer.addEventListener( "ended", function() { let nextTrack = (currentTrack + 1) % tracks.length; selectTrack( nextTrack + 1, true ); } ); /* ================================================== 페이지 처음 열었을 때 1번 곡과 가사 표시 ================================================== */ selectTrack(1, false);
