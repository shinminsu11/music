    <script>
        const dotKey = document.getElementById('dotKey');
        const output = document.getElementById('output');
        const mainKeyboard = document.getElementById('mainKeyboard');
        const symbolBoard = document.getElementById('symbolBoard');
        const clipboardBoard = document.getElementById('clipboardBoard');

        let startX = 0;
        let startY = 0;

        dotKey.addEventListener('touchstart', (e) => {
            startX = e.touches[0].clientX;
            startY = e.touches[0].clientY;
            dotKey.style.background = '#555';
        });

        dotKey.addEventListener('touchend', (e) => {
            dotKey.style.background = '#363636';
            let endX = e.changedTouches[0].clientX;
            let endY = e.changedTouches[0].clientY;
            let diffX = endX - startX;
            let diffY = endY - startY;
            let threshold = 30;

            if (Math.abs(diffX) > Math.abs(diffY)) {
                if (Math.abs(diffX) > threshold) {
                    if (diffX > 0) { output.value += 'ㅏ'; } 
                    else { output.value += 'ㅓ'; }
                }
            } else {
                if (Math.abs(diffY) > threshold) {
                    if (diffY > 0) { output.value += 'ㅜ'; } 
                    else { output.value += 'ㅗ'; }
                }
            }
            output.focus();
        });

        function switchBoard(type) {
            mainKeyboard.classList.add('hidden');
            symbolBoard.classList.add('hidden');
            clipboardBoard.classList.add('hidden');

            if (type === 'main') {
                mainKeyboard.classList.remove('hidden');
            } else if (type === 'symbol') {
                symbolBoard.classList.remove('hidden');
            } else if (type === 'clipboard') {
                clipboardBoard.classList.remove('hidden');
            }
            output.focus();
        }

        function typeKey(char) {
            output.value += char;
            output.focus();
        }

        function backspaceKey() {
            output.value = output.value.slice(0, -1);
            output.focus();
        }

        // 클립보드에 복사된 내용을 읽어와서 버튼으로 추가하는 함수
        async function fetchClipboard() {
            try {
                // 브라우저 클립보드 권한 허용 창이 뜰 수 있습니다
                const text = await navigator.clipboard.readText();
                if (text) {
                    const clipBoardDiv = document.getElementById('clipboardBoard');
                    const newBtn = document.createElement('div');
                    newBtn.className = 'key clip-item';
                    newBtn.innerText = text;
                    newBtn.onclick = () => typeKey(text);
                    // 맨 위 '가져오기' 버튼 바로 아래에 추가
                    clipBoardDiv.insertBefore(newBtn, clipBoardDiv.children[1]);
                } else {
                    alert("클립보드에 복사된 내용이 없습니다.");
                }
            } catch (err) {
                alert("클립보드 접근 권한이 거부되었거나 지원하지 않는 환경입니다.");
            }
        }
    </script>
